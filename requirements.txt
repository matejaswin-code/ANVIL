# --- mesh handling (postprocess, mock backend, stats, format conversion) ---
trimesh==4.5.3
# Lower bounds rather than exact pins: the optional real backends (rembg for
# Hunyuan3D's background removal, diffusers for image generation) require newer
# releases, and an exact pin here makes re-running setup silently downgrade them.
numpy>=2.2.1
pillow>=11.1.0

# Backs trimesh's decimation. Without it trimesh raises on
# simplify_quadric_decimation and every style's target triangle count is
# silently ignored — which makes the presets look identical. Not optional.
# 0.1.13 is the first release with a prebuilt abi3 wheel for Python 3.13;
# earlier pins compile from source and need MSVC on Windows.
fast-simplification==0.1.13

# Voxelisation and mesh boolean support used by the blocky presets
manifold3d==3.0.1

# trimesh treats scipy as optional, but the postprocess chain hits code paths
# that import it — without it the pipeline fails at the smoothing step.
scipy

# --- optional but small: speeds up trimesh's GLB/GLTF path ---
pycollada==0.8
